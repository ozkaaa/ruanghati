Playbook 1 — check.yml
cat > /etc/ansible/playbooks/check.yml << 'EOF'
---
- name: Checking Playbook
  hosts: windows
  tasks:
    - name: Ensure C:\ITNSA directory exists
      win_file:
        path: C:\ITNSA
        state: directory

    - name: Create ansible_ready.txt with content
      win_copy:
        content: "Ansible Connected"
        dest: C:\ITNSA\ansible_ready.txt
EOF

ansible-playbook /etc/ansible/playbooks/check.yml


2. deploy_project.yml
Buat dulu file variabelnya:
bashcat > /etc/ansible/itnsa_project.json << 'EOF'
{
  "project_name": "ProjectA",
  "full_access": "ITNSA\\Administrator",
  "read_only": "ITNSA\\Domain Users"
}
EOF

cat > /etc/ansible/playbooks/deploy_project.yml << 'EOF'
---
- name: Deploy Shared Folder Project
  hosts: SRV.itnsa.id
  vars_files:
    - /etc/ansible/itnsa_project.json
  tasks:
    - name: Create project directory
      win_file:
        path: "C:\\Shared Folder\\{{ project_name }}"
        state: directory

    - name: Create shared folder
      win_share:
        name: "{{ project_name }}"
        path: "C:\\Shared Folder\\{{ project_name }}"
        full: "{{ full_access }}"
        read: "{{ read_only }}"
        state: present
EOF

ansible-playbook /etc/ansible/playbooks/deploy_project.yml

deploy_website.yml
cat > /etc/ansible/playbooks/deploy_website.yml << 'EOF'
---
- name: Deploy Website on DC.lab.itnsa.id
  hosts: DC.lab.itnsa.id
  tasks:
    - name: Create DNS A record for site
      win_shell: |
        Add-DnsServerResourceRecordA `
          -ZoneName "lab.itnsa.id" `
          -Name "{{ site_name.split('.')[0] }}" `
          -IPv4Address "10.1.1.10" `
          -AllowUpdateAny
      ignore_errors: yes

    - name: Create website root directory
      win_file:
        path: "C:\\project_website\\{{ site_name }}"
        state: directory

    - name: Create index.html with site_name content
      win_copy:
        content: "{{ site_name }}"
        dest: "C:\\project_website\\{{ site_name }}\\index.html"

    - name: Create IIS website
      win_iis_website:
        name: "{{ site_name }}"
        physical_path: "C:\\project_website\\{{ site_name }}"
        hostname: "{{ site_name }}"
        port: 80
        state: started
EOF

ansible-playbook /etc/ansible/playbooks/deploy_website.yml \
  -e site_name=test.lab.itnsa.id
